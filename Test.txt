 * Уменьшить количество товара на 1.
 *
 * Если осталась одна штука —
 * полностью удаляем позицию.
 */
public static function decreaseItem(
    $userId,
    $productId,
    $sizeId
) {
    $cart = self::getOrCreateByUserId($userId);

    if (!$cart) {
        return false;
    }

    $db = Database::connect();


    // Получаем текущее количество.
    $stmt = $db->prepare("
        SELECT quantity
        FROM cart_items

        WHERE cart_id = :cart_id
          AND product_id = :product_id
          AND size_id = :size_id

        LIMIT 1
    ");

    $stmt->execute([
        'cart_id' => $cart['id'],
        'product_id' => $productId,
        'size_id' => $sizeId
    ]);

    $item = $stmt->fetch(PDO::FETCH_ASSOC);


    if (!$item) {
        return false;
    }


    // Если осталась одна штука —
    // удаляем позицию полностью.
    if ((int) $item['quantity'] <= 1) {

        $stmt = $db->prepare("
            DELETE FROM cart_items

            WHERE cart_id = :cart_id
              AND product_id = :product_id
              AND size_id = :size_id
        ");

    } else {

        // Иначе уменьшаем количество на 1.
        $stmt = $db->prepare("
            UPDATE cart_items

            SET quantity = quantity - 1

            WHERE cart_id = :cart_id
              AND product_id = :product_id
              AND size_id = :size_id
        ");
    }


    return $stmt->execute([
        'cart_id' => $cart['id'],
        'product_id' => $productId,
        'size_id' => $sizeId
    ]);
else {

        // Иначе уменьшаем количество на 1.

            UPDATE cart_items

            SET quantity = quantity - 1

            WHERE cart_id = :cart_id

              AND size_id = :size_id

    }

        'cart_id' => $cart['id'],
        'product_id' => $productId,
        'size_id' => $sizeId
    ]);
}
